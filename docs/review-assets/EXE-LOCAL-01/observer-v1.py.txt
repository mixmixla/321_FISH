def job_pids(proc):
    class List(ctypes.Structure):
        _fields_=[('assigned',wintypes.DWORD),('count',wintypes.DWORD),('ids',ctypes.c_size_t*512)]
    x=List()
    assert jobs._kernel().QueryInformationJobObject(proc._job_handle,3,ctypes.byref(x),ctypes.sizeof(x),None)
    assert x.count<=512
    return [int(v) for v in x.ids[:x.count]]

def zero_job(proc,timeout=5):
    deadline=time.monotonic()+timeout
    while proc.active_count() and time.monotonic()<deadline:time.sleep(.05)
    return proc.active_count()==0
def cleanup(proc,row):
    row['before_cleanup']={'poll':proc.poll(),'job_active':proc.active_count(),'job_pids':job_pids(proc)}
    row['cleanup_forced']=bool(proc.active_count())
    if row['cleanup_forced']:row['cleanup_ok']=bool(proc.terminate_tree())
    else:row['cleanup_ok']=True
    row['after_cleanup']={'poll':proc.poll(),'job_active':proc.active_count(),'job_pids':job_pids(proc)}
    assert row['cleanup_ok'] and row['after_cleanup']['job_active']==0
    proc.close()

def windows(proc):
    members=set(job_pids(proc));found=[]
    @Callback
    def callback(hwnd,lparam):
        pid=wintypes.DWORD();user32.GetWindowThreadProcessId(hwnd,ctypes.byref(pid))
        if pid.value in members:
            title=ctypes.create_unicode_buffer(512);klass=ctypes.create_unicode_buffer(128)
            user32.GetWindowTextW(hwnd,title,512);user32.GetClassNameW(hwnd,klass,128)
            found.append({'hwnd':int(hwnd),'pid':int(pid.value),'title':title.value,'class':klass.value,'visible':bool(user32.IsWindowVisible(hwnd)),'image':process_image_path(pid.value)})
        return True
    assert proc._desktop_handle and user32.EnumDesktopWindows(proc._desktop_handle,callback,0)
    return found
